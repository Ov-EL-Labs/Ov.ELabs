// GSAP Left Bar Choreography (Kinetic Headline + Spotlight Logos + Odometer + Sparklines)
import gsap from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
gsap.registerPlugin(ScrollTrigger);

// 1. Staggered 3D Hero Word Reveal
gsap.from('.hero-word', {
  y: 36,
  opacity: 0,
  rotationX: -45,
  transformPerspective: 800,
  duration: 0.75,
  stagger: 0.045,
  ease: 'expo.out'
});

// 2. Mouse-Tracking Radial Spotlight on the 8 Client Logo Cells
document.querySelectorAll('.logo-cell').forEach((cell) => {
  cell.addEventListener('mousemove', (e) => {
    const r = cell.getBoundingClientRect();
    cell.style.setProperty('--cell-mx', `${e.clientX - r.left}px`);
    cell.style.setProperty('--cell-my', `${e.clientY - r.top}px`);
  });
});

// 3. Odometer Count-Up + SVG Sparkline Draw for 50% & 60% Metrics
['#metric-leads', '#metric-engagement'].forEach((selector) => {
  const el = document.querySelector(selector);
  const target = parseInt(el.dataset.target, 10);
  const obj = { val: 0 };
  gsap.to(obj, {
    val: target,
    duration: 1.6,
    ease: 'power3.out',
    onUpdate: () => { el.textContent = Math.round(obj.val); }
  });
});

// 4. Continuous Laser Photon Sweep on Metric Divider Lines
gsap.to('.laser-photon', {
  left: '105%',
  duration: 2.4,
  repeat: -1,
  ease: 'power2.inOut',
  stagger: 1.1
});
